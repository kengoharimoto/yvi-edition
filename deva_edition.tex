% Devanagari MT's interword space has very little stretch and long
% compounds can only break between akṣaras, so at the default tolerance
% TeX often finds no acceptable break and sets an overfull line instead.
% Accept looser lines rather than lines that run into the margin.
\tolerance=1000
\emergencystretch=1.5em

% In the Devanagari version the kundala is the Devanagari abbreviation sign.
\renewcommand{\defkundala}{\renewcommand{\kundala}{॰}}
% Under current polyglossia, \rmfamily inside the sanskrit environment stays
% the Devanagari font, so \cm must select the Latin family explicitly.
\renewcommand{\cm}{\latinroman}
% Apparatus: readings and lemmata in Devanagari, sigla and framework in Latin.
\renewcommand{\rdg}[2]{{\sanskritfont #1}\ {\cm #2}}
\renewcommand*{\Anotefmt}{%
  \renewcommand{\lemmafmt}[1]{{\sanskritfont\kundala ##1}\thinspace]\,}%
  \renewcommand{\notefmt}[1]{\cm\spaceskip 0pt \xspaceskip 0pt ##1}%
}
% Editorial notes (\edn) are English prose: Latin font throughout.
\renewcommand*{\Enotefmt}{%
  \renewcommand*{\linesfmt}[1]{\kern -2.0em {\latinroman\textbf{##1}}\enspace}%
  \renewcommand{\lemmafmt}[1]{}%
  \renewcommand{\notefmt}[1]{\latinroman ##1}%
}
% Symbols that must not be looked up in the Devanagari font:
\renewcommand{\om}{{\cm\emph{om.}}}
\renewcommand{\dag}{{\cm\textdagger}}
\renewcommand{\ins}{\il{}}
\renewcommand{\ine}{\ir{}}
\renewcommand*{\lemmaellipsis}{{\cm\,\ldots\,}}
% Lemma-ellipsis kundalas in Devanagari:
\renewcommand*{\varsm}[2]{\Anote{#1\<<॰>\>}{#2}}
\renewcommand*{\vrtsm}[3]{\Anote{#1\<<॰>\>}{{\cm #2, } #3}\hyb}
\renewcommand*{\vrtmm}[3]{\hyb\defkundala\Anote{#1\<<॰>\>}{{\cm #2, } #3}\undefkundala\hyb}
% The C register (\tst, \donote on \tstwll labels) holds Sanskrit
% parallels: Devanagari font, with Latin bits marked \cm in the source.
\makeatletter
\renewcommand*{\Cnotefmt}{%
  \renewcommand*{\linesfmt}[1]{\kern -2.0em {\latinroman\textbf{##1}}\enspace}%
  \renewcommand{\lemmafmt}[1]{%
    \def\@inlemma{YES}%
    \let\Large=\normalsize
    {\sanskritfont\kundala ##1}\thinspace]\enskip
    \def\@inlemma{NO}}%
  \renewcommand{\notefmt}[1]{\sanskritfont\spaceskip 0pt \xspaceskip 0pt ##1}%
}
\makeatother
% The bottom (Bhāṣya) register is pure Devanagari.
\renewcommand*{\Bnotefmt}{%
  \renewcommand{\lemmafmt}[1]{\enskip}%
  \renewcommand{\notefmt}[1]{\sanskritfont\spaceskip 0pt \xspaceskip 0pt ##1}%
}
